function convertAndCopy() { try { const rawText = document.getElementById('inputText').value; const msgEl = document.getElementById('outputMessage'); const previewEl = document.getElementById('jsonPreview'); if (!rawText) { alert('テキストを入力してください。'); return; } // --- データの抽出 --- const nameMatch = rawText.match(/キャラクター名[：\s]*(.+)/) || rawText.match(/名前[：\s]*(.+)/); const dollName = nameMatch ? nameMatch[1].trim() : '名無しのドール'; const posMatch = rawText.match(/ポジション[：\s]*(.+)/); const position = posMatch ? posMatch[1].trim() : '未設定'; const classMatch = rawText.match(/クラス[：\s]*(.+)/); const charClass = classMatch ? classMatch[1].trim() : ''; const initMatch = rawText.match(/行動値[：\s]*(\d+)/); const initiative = initMatch ? parseInt(initMatch[1], 10) : 0; // --- ココフォリア用メモの整形 --- // ご希望の形式や、キャラシの情報を綺麗にまとめます const memoText = `【${dollName}】\nポジション: ${position} / クラス: ${charClass}\n行動値: ${initiative}\n\n` + `--- パーソナルデータ ---\n` + (rawText.match(/暗示[\s\S]*?(?=\n■|\n\n■)/) || [''])[0] + '\n\n' + `--- マニューバ・パーツ一覧 ---\n` + (rawText.match(/\[部位\][\s\S]*?(?=\n■|\n\n$)/) || [''])[0] + '\n\n' + `--- キャラクターシート全文 ---\n${rawText}`; // --- ココフォリア用JSON構造 --- const cocofoliaData = { "kind": "character", "data": { "name": dollName, "memo": memoText, "initiative": initiative, "params": [ { "label": "狂気度", "value": 0, "max": 10 }, { "label": "ドゥーム", "value": 0, "max": 5 } ], "status": [ { "label": "頭部", "value": 2, "max": 2 }, { "label": "胴体", "value": 3, "max": 3 }, { "label": "腕部", "value": 2, "max": 2 }, { "label": "脚部", "value": 2, "max": 2 } ], "chatPalette": "1+1 判定\n2d6 狂気判定\n5>=[攻撃力] 攻撃" } }; const jsonString = JSON.stringify(cocofoliaData, null, 2); // プレビューに表示 previewEl.textContent = jsonString; // クリップボードへコピー if (navigator.clipboard && navigator.clipboard.writeText) { navigator.clipboard.writeText(jsonString).then(() => { msgEl.textContent = 'コピー完了しました！ココフォリアの部屋で [Ctrl + V] で配置してください。'; }).catch(err => { msgEl.textContent = 'プレビュー欄から手動でコピーしてください。'; }); } else { msgEl.textContent = '自動コピー非対応です。下のプレビューからコピーしてください。'; } } catch (error) { alert('エラーが発生しました: ' + error.message); } }
